"""Stage 03 acceptance demo: one run logs and records audit events; tamper → verify False."""
import io, json, sys, tempfile
from datetime import date
from pathlib import Path
from atlas.core.audit import AuditLog, EventType, audit_path
from atlas.core.config import AtlasConfig
from atlas.core.logging import configure_logging, get_logger, reset_logging
from atlas.core.run_context import run_context, run_dir

lake = Path(tempfile.mkdtemp())
log = get_logger("atlas.demo")
with run_context(config=AtlasConfig(profile="backtest"), as_of=date(2024, 1, 2)) as ctx:
    configure_logging(log_file=run_dir(lake, ctx.run_id) / "atlas.log", console=io.StringIO())
    audit = AuditLog(audit_path(lake, ctx.run_id))
    audit.record(EventType.RUN_STARTED, "fund", {"profile": ctx.mode})
    log.info("run started")
    for day in (date(2024, 1, 2), date(2024, 1, 3)):
        with run_context(as_of=day, stage="risk"):
            event = audit.record(EventType.RISK_DECISION, "risk_engine", {"approved": True})
            log.info("risk decision", event_id=event.event_id)
    audit.record(EventType.RUN_FINISHED, "fund", {})
    log.info("run finished")
reset_logging()
print("run_id:", ctx.run_id, "| mode:", ctx.mode, "| config_fingerprint:", ctx.config_fingerprint)
lines = (run_dir(lake, ctx.run_id) / "atlas.log").read_text().splitlines()
print(f"log lines: {len(lines)}; every line carries this run_id:",
      all(json.loads(l)["run_id"] == ctx.run_id for l in lines))
for l in lines: print("  ", l)
path = audit_path(lake, ctx.run_id)
events = path.read_text().splitlines()
print(f"audit events: {len(events)} at runs/{ctx.run_id}/audit.jsonl")
for e in events:
    d = json.loads(e); print(f"   {d['event_type']:<14} as_of={d['as_of']} prev={d['prev_hash'][:12]} hash={d['hash'][:12]}")
print("verify_chain (untouched):", AuditLog.verify_chain(path))
path.write_text(path.read_text().replace('"approved":true', '"approved":false', 1))
print("verify_chain (one payload edited):", AuditLog.verify_chain(path))
